#include <tunables/global>

profile lovelacer flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/nameservice>
  #include <abstractions/openssl>

  capability net_bind_service,
  capability dac_override,
  capability setgid,
  capability setuid,

  network inet stream,
  network inet6 stream,
  network unix stream,

  # Alpine: /bin/sh is a symlink to /bin/busybox; the kernel resolves
  # the symlink before AppArmor's exec check, so we must whitelist
  # /bin/busybox as well as /bin/sh.
  /bin/sh ix,
  /bin/busybox ix,
  /usr/bin/node ix,
  /usr/bin/jq ix,

  /run.sh r,
  # `m` (mmap with PROT_EXEC) is required for Node's dlopen of native
  # `.node` modules such as better-sqlite3 — `r` alone fails with
  # ERR_DLOPEN_FAILED ("Permission denied"). The whole tree under /app
  # is our own application code, so granting mmap-execute is fine.
  /app/** rm,
  # SQLite needs advisory file locks on its database and journal sidecars.
  /data/** rwk,
  /tmp/** rwk,
  /proc/sys/kernel/random/uuid r,
  /proc/*/status r,
  /proc/*/stat r,
  /proc/*/maps r,
  /proc/*/cmdline r,
  /proc/self/exe r,
  /proc/self/maps r,
  /proc/self/fd/ r,
  /proc/self/fd/* r,
  /etc/passwd r,
  /etc/group r,
  /etc/nsswitch.conf r,
  /etc/resolv.conf r,
  /etc/ssl/** r,
  /etc/hosts r,
}
